// (Integer).to_s(16)
static void c_integer_to_s(struct VM *vm, mrbc_value v[], int argc)
{
mrbc_value *recv = &v[1]; // expect Integer.
mrbc_value arg1 = mrbc_integer_value(16);
mrbc_value ret = mrbc_send( vm, v, argc, recv, "to_s", 1, &arg1 );
SET_RETURN(ret);
