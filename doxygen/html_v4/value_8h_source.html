546static inline void mrbc_incref(mrbc_value *v)
547{
548 if( mrbc_type(*v) <= MRBC_TT_INC_DEC_THRESHOLD ) return;
549
550 assert( v->obj->ref_count != 0 );
551 assert( v->obj->ref_count != 0xffff ); // check max value.
552 v->obj->ref_count++;
553}
554
555
556//================================================================
561static inline void mrbc_decref(mrbc_value *v)
562{
563 if( mrbc_type(*v) <= MRBC_TT_INC_DEC_THRESHOLD ) return;
564
565 assert( v->obj->ref_count != 0 );
566 assert( v->obj->ref_count != 0xffff ); // check broken data.
567
568 if( --v->obj->ref_count != 0 ) return;
569
570 (*mrbc_delfunc[v->tt])(v);
571}

603static inline uint16_t bin_to_uint16( const void *s )
604{
605#if defined(MRBC_LITTLE_ENDIAN) && !defined(MRBC_REQUIRE_32BIT_ALIGNMENT)
606 // Little endian, no alignment.
607 uint16_t x = *((uint16_t *)s);
608 x = (x << 8) | (x >> 8);
609
610#elif defined(MRBC_BIG_ENDIAN) && !defined(MRBC_REQUIRE_32BIT_ALIGNMENT)
611 // Big endian, no alignment.
612 uint16_t x = *((uint16_t *)s);
613
614#elif defined(MRBC_REQUIRE_32BIT_ALIGNMENT)
615 // 32bit alignment required.
616 uint8_t *p = (uint8_t *)s;
617 uint16_t x = *p++;
618 x <<= 8;
619 x |= *p;
620
621#else
622 #error "Specify MRBC_BIG_ENDIAN or MRBC_LITTLE_ENDIAN"
623#endif
624
625 return x;
626}

635static inline uint32_t bin_to_uint32( const void *s )
636{
637#if defined(MRBC_LITTLE_ENDIAN) && !defined(MRBC_REQUIRE_32BIT_ALIGNMENT)
638 // Little endian, no alignment.
639 uint32_t x = *((uint32_t *)s);
640 x = (x << 24) | ((x & 0xff00) << 8) | ((x >> 8) & 0xff00) | (x >> 24);
641
642#elif defined(MRBC_BIG_ENDIAN) && !defined(MRBC_REQUIRE_32BIT_ALIGNMENT)
643 // Big endian, no alignment.
644 uint32_t x = *((uint32_t *)s);
645
646#elif defined(MRBC_REQUIRE_32BIT_ALIGNMENT)
647 // 32bit alignment required.
648 uint8_t *p = (uint8_t *)s;
649 uint32_t x = *p++;
650 x <<= 8;
651 x |= *p++;
652 x <<= 8;
653 x |= *p++;
654 x <<= 8;
655 x |= *p;
656
657#endif
658
659 return x;
660}

721static inline double bin_to_double64( const void *s )
722{
723#if defined(MRBC_LITTLE_ENDIAN) && !defined(MRBC_REQUIRE_64BIT_ALIGNMENT)
724 // Little endian, no alignment.
725 return *((double *)s);
726
727#elif defined(MRBC_BIG_ENDIAN) && !defined(MRBC_REQUIRE_64BIT_ALIGNMENT)
728 // Big endian, no alignment.
729 double x;
730 uint8_t *p1 = (uint8_t*)s;
731 uint8_t *p2 = (uint8_t*)&x + 7;
732 int i;
733 for( i = 7; i >= 0; i-- ) {
734 *p2-- = *p1++;
735 }
736 return x;
737
