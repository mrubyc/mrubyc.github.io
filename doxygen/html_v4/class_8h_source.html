262static inline mrbc_class *find_class_by_object(const mrbc_value *obj)
263{
264 assert( mrbc_type(*obj) >= 0 );
265 assert( mrbc_type(*obj) <= MRBC_TT_MAXVAL );
266
267 mrbc_class *cls = mrbc_class_tbl[ mrbc_type(*obj) ];
268 if( !cls ) {
269 switch( mrbc_type(*obj) ) {
270 case MRBC_TT_CLASS: // fall through.
271 case MRBC_TT_MODULE: cls = obj->cls; break;
272 case MRBC_TT_OBJECT: cls = obj->instance->cls; break;
273 case MRBC_TT_EXCEPTION: cls = obj->exception->cls; break;
274 default:
275 assert(!"Invalid value type.");
276 }
277 }
278
279 return cls;
280}
