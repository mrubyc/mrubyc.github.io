82static int binary_search(mrbc_kv_handle *kvh, mrbc_sym sym_id)
83{
84 int left = 0;
85 int right = kvh->n_stored - 1;
86 if( right < 0 ) return -1;
87
88 while( left < right ) {
89 int mid = (left + right) / 2;
90 if( kvh->data[mid].sym_id < sym_id ) {
91 left = mid + 1;
92 } else {
93 right = mid;
94 }
95 }
96
97 return left;
98}

228int mrbc_kv_set(mrbc_kv_handle *kvh, mrbc_sym sym_id, mrbc_value *set_val)
229{
230 int idx = binary_search(kvh, sym_id);
231 if( idx < 0 ) {
232 idx = 0;
233 goto INSERT_VALUE;
234 }
235
236 // replace value ?
237 if( kvh->data[idx].sym_id == sym_id ) {
238 mrbc_decref( &kvh->data[idx].value );
239 kvh->data[idx].value = *set_val;
240 return 0;
241 }
242
243 if( kvh->data[idx].sym_id < sym_id ) {
244 idx++;
245 }
246
247 INSERT_VALUE:
248 // need alloc?
249 if( kvh->data_size == 0 ) {
250 kvh->data = mrbc_alloc(kvh->vm, sizeof(mrbc_kv) * MRBC_KV_SIZE_INIT);
251 kvh->data_size = MRBC_KV_SIZE_INIT;
252
253#if defined(MRBC_DEBUG)
254 memcpy( kvh->data->obj_mark_, "KV", 2 );
255#endif
256
257 // need resize?
258 } else if( kvh->n_stored >= kvh->data_size ) {
259 mrbc_kv_resize(kvh, kvh->data_size + MRBC_KV_SIZE_INCREMENT);
260 }
261
262 // need move data?
263 if( idx < kvh->n_stored ) {
264 int size = sizeof(mrbc_kv) * (kvh->n_stored - idx);
265 memmove( &kvh->data[idx+1], &kvh->data[idx], size );
266 }
267
268 kvh->data[idx].sym_id = sym_id;
269 kvh->data[idx].value = *set_val;
270 kvh->n_stored++;
271
272 return 0;
273}

357int mrbc_kv_remove(mrbc_kv_handle *kvh, mrbc_sym sym_id)
358{
359 int idx = binary_search(kvh, sym_id);
360 if( idx < 0 ) return 0;

362

364 kvh->n_stored--;

367
368 return 0;
