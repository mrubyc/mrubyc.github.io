114static void c_range_equal3(mrbc_vm *vm, mrbc_value v[], int argc)
115{
116 if( mrbc_type(v[0]) == MRBC_TT_CLASS ) {
117 mrbc_value result = mrbc_send( vm, v, argc, &v[1], "kind_of?", 1, &v[0] );
118 SET_RETURN( result );
119 return;
120 }
121
122 int cmp_first = mrbc_compare( &v[0].range->first, &v[1] );
123 int result = (cmp_first <= 0);
124 if( !result ) goto DONE;
125

127 result = (v->range->flag_exclude) ? (cmp_last < 0) : (cmp_last <= 0);
128
129 DONE:
130 SET_BOOL_RETURN( result );
131}
