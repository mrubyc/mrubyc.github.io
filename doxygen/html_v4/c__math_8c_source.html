37static double to_double( mrbc_vm *vm, const mrbc_value *v )
38{
39 switch( mrbc_type(*v) ) {
40 case MRBC_TT_INTEGER: return (double)v->i;
41 case MRBC_TT_FLOAT: return (double)v->d;
42 default: break;
43 }
44
45 mrbc_raise(vm, MRBC_CLASS(TypeError), 0);
46 return 0.0;
47}

164static void c_math_ldexp(mrbc_vm *vm, mrbc_value v[], int argc)
165{
166 int exp;
167 switch( mrbc_type(v[2]) ) {
168 case MRBC_TT_INTEGER: exp = v[2].i; break;
169 case MRBC_TT_FLOAT: exp = (int)v[2].d; break;
170 default:
171 mrbc_raise(vm, MRBC_CLASS(TypeError), 0);
172 return;
173 }
174
175 v[0] = mrbc_float_value(vm, ldexp( to_double(vm, &v[1]), exp ));
176}
