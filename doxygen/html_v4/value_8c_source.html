68int mrbc_compare(const mrbc_value *v1, const mrbc_value *v2)
69{
70#if MRBC_USE_FLOAT
71 mrbc_float_t d1, d2;
72#endif
73
74 // if TT_XXX is different
75 if( mrbc_type(*v1) != mrbc_type(*v2) ) {
76#if MRBC_USE_FLOAT
77 // but Numeric?
78 if( mrbc_type(*v1) == MRBC_TT_INTEGER && mrbc_type(*v2) == MRBC_TT_FLOAT ) {
79 d1 = v1->i;
80 d2 = v2->d;
81 goto CMP_FLOAT;
82 }
83 if( mrbc_type(*v1) == MRBC_TT_FLOAT && mrbc_type(*v2) == MRBC_TT_INTEGER ) {
84 d1 = v1->d;
85 d2 = v2->i;
86 goto CMP_FLOAT;
87 }
88#endif
89
90 // leak Empty?
91 if((mrbc_type(*v1) == MRBC_TT_EMPTY && mrbc_type(*v2) == MRBC_TT_NIL) ||
92 (mrbc_type(*v1) == MRBC_TT_NIL && mrbc_type(*v2) == MRBC_TT_EMPTY)) return 0;
93
94 // other case
95 return mrbc_type(*v1) - mrbc_type(*v2);
96 }
97
98 // check value
99 switch( mrbc_type(*v1) ) {
100 case MRBC_TT_NIL:
101 case MRBC_TT_FALSE:
102 case MRBC_TT_TRUE:
103 return 0;
104
105 case MRBC_TT_INTEGER:
106 return mrbc_integer(*v1) - mrbc_integer(*v2);
107
108 case MRBC_TT_SYMBOL: {
109 const char *str1 = mrbc_symid_to_str(mrbc_symbol(*v1));
110 const char *str2 = mrbc_symid_to_str(mrbc_symbol(*v2));
111 int diff = strlen(str1) - strlen(str2);
112 int len = diff < 0 ? strlen(str1) : strlen(str2);
113 int res = memcmp(str1, str2, len);
114 return (res != 0) ? res : diff;
115 }
116
117#if MRBC_USE_FLOAT
118 case MRBC_TT_FLOAT:
119 d1 = mrbc_float(*v1);
120 d2 = mrbc_float(*v2);
121 goto CMP_FLOAT;
122#endif
123
124 case MRBC_TT_CLASS:
125 case MRBC_TT_MODULE:
126 case MRBC_TT_OBJECT:
127 case MRBC_TT_PROC:
128 return (v1->cls > v2->cls) * 2 - (v1->cls != v2->cls);
129
130 case MRBC_TT_ARRAY:
131 return mrbc_array_compare( v1, v2 );
132
133#if MRBC_USE_STRING
134 case MRBC_TT_STRING:
135 return mrbc_string_compare( v1, v2 );
136#endif
137
138 case MRBC_TT_RANGE:
139 return mrbc_range_compare( v1, v2 );
140
141 case MRBC_TT_HASH:
142 return mrbc_hash_compare( v1, v2 );
143
144 default:
145 return 1;
146 }
147
148#if MRBC_USE_FLOAT
149 CMP_FLOAT:
150 return -1 + (d1 == d2) + (d1 > d2)*2; // caution: NaN == NaN is false
151#endif
152}

188mrbc_int_t mrbc_atoi( const char *s, int base )
189{
190 int ret = 0;
191 int sign = 0;
192
193 REDO:
194 switch( *s ) {
195 case '-':
196 sign = 1;
197 // fall through.
198 case '+':
199 s++;
200 break;
201
202 case ' ':
203 s++;
204 goto REDO;
205 }
206
207 int ch;
208 while( (ch = *s++) != '\0' ) {
209 int n;
210
211 if( 'a' <= ch ) {
212 n = ch - 'a' + 10;
213 } else
214 if( 'A' <= ch ) {
215 n = ch - 'A' + 10;
216 } else
217 if( '0' <= ch && ch <= '9' ) {
218 n = ch - '0';
219 } else {
220 break;
221 }
222 if( n >= base ) break;
223
224 ret = ret * base + n;
225 }
226
227 if( sign ) ret = -ret;
228
229 return ret;
230}

241int mrbc_strcpy( char *dest, int destsize, const char *src )
242{
243 int n = destsize;
244 if( n <= 0 ) return 0;
245
246 while( --n != 0 ) {
247 if( (*dest++ = *src++) == 0 ) goto RETURN;
248 }
249 *dest = 0;
250
251 RETURN:
252 return destsize - n - 1;
253}
